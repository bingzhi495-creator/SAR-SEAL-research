% \chipview{x}{y}{width in cm}{flip: 0=id, 1=hflip} — dark SAR chip, bright ships
\newcommand{\chipview}[4]{%
  \begin{scope}[shift={(#1,#2)}]
    \pgfmathsetmacro{\hw}{#3/2}
    \fill[black!75] (-\hw,-\hw) rectangle (\hw,\hw);
    \draw[black!40, line width=0.4pt] (-\hw,-\hw) rectangle (\hw,\hw);
    \pgfmathsetmacro{\ax}{#4*0.36*#3-0.18*#3}
    \pgfmathsetmacro{\aay}{0.12*#3}
    \pgfmathsetmacro{\rot}{35-#4*70}
    \fill[white, rotate around={\rot:(\ax,\aay)}]
      (\ax,\aay) ellipse [x radius=0.17*#3, y radius=0.05*#3];
    \pgfmathsetmacro{\bx}{0.22*#3-#4*0.44*#3}
    \fill[white] (\bx,-0.20*#3) ellipse [x radius=0.09*#3, y radius=0.035*#3];
  \end{scope}}
