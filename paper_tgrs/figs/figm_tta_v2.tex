% --- \gridchip{x}{y}{side}{step} : the SAME hull sampled on a coarse or a
%     fine pixel lattice (figure-local illustration for the inset).
\newcommand{\gridchip}[4]{%
  \begin{scope}[shift={(#1,#2)}]
    \pgfmathsetmacro{\gw}{#3/2}
    \pgfmathsetmacro{\ga}{0.30*#3}
    \pgfmathsetmacro{\gb}{0.075*#3}
    \fill[sarink] (-\gw,-\gw) rectangle (\gw,\gw);
    \begin{scope}
      \clip (-\gw,-\gw) rectangle (\gw,\gw);
      \fill[white, opacity=0.90, rotate=24] (0,0) ellipse[x radius=\ga, y radius=\gb];
      \draw[white, opacity=0.32, line width=0.3pt, step=#4] (-\gw,-\gw) grid (\gw,\gw);
    \end{scope}
    \draw[black!55, line width=0.4pt] (-\gw,-\gw) rectangle (\gw,\gw);
  \end{scope}}
